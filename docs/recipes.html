source data

SELECT col->'foo'      -- NULL if absent

SELECT doc->'$.foo'
duckdb -c "SELECT doc->'$.foo'"

doc.get("foo")        # None
// a missing field is simply absent from the result
mongosh --eval '// a missing field is simply absent from the result'
.foo?                 # no error even on a non-object
jq '.foo? # no error even on a non-object'
→
{}   (foo is MISSING → the key is omitted)
